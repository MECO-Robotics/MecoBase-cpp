cmake_minimum_required(VERSION 3.24)
project(FrcJolt LANGUAGES CXX)
include(FetchContent)
foreach(option OVERRIDE_CXX_FLAGS INTERPROCEDURAL_OPTIMIZATION USE_STATIC_MSVC_RUNTIME_LIBRARY USE_SSE4_1 USE_SSE4_2 USE_AVX USE_AVX2 USE_AVX512 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE ENABLE_ALL_WARNINGS ENABLE_INSTALL)
  set(${option} OFF CACHE BOOL "Portable embedded Jolt build" FORCE)
endforeach()
set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "Bridge uses standard exceptions" FORCE)
FetchContent_Declare(Jolt URL https://github.com/jrouwe/JoltPhysics/archive/0373ec0dd762e4bc2f6acdb08371ee84fa23c6db.tar.gz URL_HASH SHA256=834644d7e9dffee9be0220748015667649d3361611b01c7e6839a5eb58e89711 SOURCE_SUBDIR Build)
FetchContent_MakeAvailable(Jolt)
add_library(frc_jolt STATIC src/SimulationWorld.cpp src/DrivePhysics.cpp src/Arena.cpp)
target_include_directories(frc_jolt PUBLIC include)
target_compile_features(frc_jolt PUBLIC cxx_std_20)
target_link_libraries(frc_jolt PUBLIC Jolt)
set_target_properties(frc_jolt Jolt PROPERTIES POSITION_INDEPENDENT_CODE ON ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config DEBUG RELEASE)
  set_target_properties(frc_jolt Jolt PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
include(CTest)
add_executable(simulation_benchmark tests/SimulationBenchmark.cpp)
target_link_libraries(simulation_benchmark PRIVATE frc_jolt)
if(BUILD_TESTING)
 add_executable(world_test tests/WorldTest.cpp)
 target_link_libraries(world_test PRIVATE frc_jolt)
 add_test(NAME world COMMAND world_test)
 add_executable(drive_physics_test tests/DrivePhysicsTest.cpp)
 target_link_libraries(drive_physics_test PRIVATE frc_jolt)
 add_test(NAME drive_physics COMMAND drive_physics_test)
 add_executable(arena_test tests/ArenaTest.cpp)
 target_link_libraries(arena_test PRIVATE frc_jolt)
 add_test(NAME arena COMMAND arena_test)
endif()
